pub typealias u8 = .u8'builtin;

fn compute(ptr: *u8) -> !Error ?u8 {
    let raw = ptr.* + 1u8;
    let addr = raw.ref;
    let ok = raw < 10u8 and not false;
    if ok {
        addr.*.as(u8)
    } else {
        null
    }
}
